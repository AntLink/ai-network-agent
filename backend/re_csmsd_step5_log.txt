#!/bin/sh
DAEMON=/media/tci/csms/bin/csmsd.elf
NAME=csmsd
DESC="Compact Spectrum Monitoring Server"
PIDFILE=/media/tci/csms/csmsd.pid
ARGS="--daemon"

set -

test -f "$DAEMON" || exit 0

case "$1" in
  start)
        echo -n "Starting $DESC: "
        start-stop-daemon -S -p $PIDFILE -x "$DAEMON" -- $ARGS
        echo "$NAME."
        ;;
  stop)
        echo -n "Stopping $DESC: "
        start-stop-daemon -K -p $PIDFILE -s QUIT
        echo "$NAME."
        ;;
  restart)
        echo -n "Restarting $DESC: "
        start-stop-daemon -K -p $PIDFILE -s QUIT
        chkprocess="csmsd.elf"
        while [[ $chkprocess == *"csmsd.elf"* ]]
        do
          chkprocess=$(ps | grep "csmsd")
          echo "waiting for csmd to exit..."
          sleep 1
        done
        start-stop-daemon -S -p $PIDFILE -x "$DAEMON" -- $ARGS
        echo "$NAME."
        ;;
  *)
        N=/etc/init.d/$NAME
        echo "Usage: $0 {start|stop|restart}"
        exit 1
        ;;
esac

exit 0
