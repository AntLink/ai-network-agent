### /media/httpd/cgi-bin/get_firmwarefile.cgi
#!/bin/bash
if [ "$REQUEST_METHOD" = "POST" ]; then
  echo "application/octet-stream"
  TMPOUT=./newfile
  cat >$TMPOUT
 
  # Get the line count
  LINES=$(wc -l $TMPOUT | cut -d ' ' -f 1)
 
  # Remove the first four lines
  tail -$((LINES - 4)) $TMPOUT >$TMPOUT.jbi
 
  # Remove the last line
  head -$((LINES - 5)) $TMPOUT.jbi >$TMPOUT
 
  # Copy everything but the new last line to a temporary file
  head -$((LINES - 6)) $TMPOUT >$TMPOUT.jbi
 
  # Copy the new last line but remove trailing \r\n
  tail -1 $TMPOUT | perl -p -i -e 's/\r\n$//' >>$TMPOUT.jbi
fi
# stop csmsd

/etc/rc5.d/S99csmsd stop
sleep 2
