#!/bin/bash
ScriptEnabled=true
#ScriptEnable=false
FILE="/media/tci/csms/log/SpectrumMonitor.log"
chkprocess="csmsd.elf"
now=$(date)
echo -e "Starting SpecturmMonitor script... $now\n" >> $FILE
echo $ScriptEnabled
while [ $ScriptEnabled ]
do
  # check if media turned read only
  mountfile=/proc/self/mountinfo
  if grep -q "mmcblk0p1 ro" $mountfile
  then
    fsck -aw /dev/mmcblk0p1
    echo "media read-only detected.. cleaning disk and reboot..." 
    now=$(date)
    echo -e "media read-only detected..$now cleaning disk and reboot...\n" >> $FILE
    sleep 5
    reboot
  else
    echo "media r/w..."
  fi 


  chkprocess=$(ps | grep "csmsd")
  if [[ $chkprocess == *"csmsd.elf"* ]]; then
    echo "csmsd running..."
  else
    echo "detected csmsd stopped, relauch csmsd..."
    now=$(date)
    echo -e "detected csmsd stopped, relauch csmsd: $now\n" >> $FILE
    /etc/rc5.d/S99csmsd start 
  fi
  sleep 30 
done
now=$(date) 
echo "Current time : $now"
echo "csmsd monitor script exited normally"
