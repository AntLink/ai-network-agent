#!/bin/sh
BACKUP_DIR="/media/backup/cloudflared"
TARGET_DIR="/media/httpd/cgi-bin"
echo "=== Restoring cloudflared ==="
/etc/init.d/cloudflared stop 2>/dev/null
sleep 1
echo "Restoring binary..."
cp "$BACKUP_DIR/cloudflared-linux-arm" "$TARGET_DIR/cloudflared-linux-arm"
chmod +x "$TARGET_DIR/cloudflared-linux-arm"
echo "Restoring script..."
cp "$BACKUP_DIR/cloudflared-script" "$TARGET_DIR/cloudflared"
chmod +x "$TARGET_DIR/cloudflared"
echo "Restoring init.d..."
cp "$BACKUP_DIR/cloudflared-initd" /etc/init.d/cloudflared
chmod +x /etc/init.d/cloudflared
echo "Starting cloudflared..."
/etc/init.d/cloudflared start
sleep 3
echo ""
echo "=== Verify ==="
/etc/init.d/cloudflared status
ps aux | grep cloudflared
echo ""
echo "Restore complete!"

=== /media/backup/restore.sh ===
#!/bin/sh
# Restore configuration from backup
# Usage: /media/backup/restore.sh <timestamp>
# Example: /media/backup/restore.sh 20260821-185400

if [ -z "$1" ]; then
    echo "Usage: $0 <timestamp>"
    echo "Available backups:"
    ls /media/backup/configs/
    exit 1
fi

BACKUP_DIR="/media/backup/configs/$1"
if [ ! -d "$BACKUP_DIR" ]; then
    echo "Backup not found: $BACKUP_DIR"
    exit 1
fi

echo "Restoring from: $BACKUP_DIR"

# Restore files
cp $BACKUP_DIR/hostname.txt /etc/hostname 2>/dev/null
cp $BACKUP_DIR/hosts.txt /etc/hosts 2>/dev/null
cp $BACKUP_DIR/resolv.conf.txt /etc/resolv.conf 2>/dev/null
cp $BACKUP_DIR/interfaces.txt /etc/network/interfaces 2>/dev/null
cp $BACKUP_DIR/fstab.txt /etc/fstab 2>/dev/null
cp $BACKUP_DIR/inittab.txt /etc/inittab 2>/dev/null
cp $BACKUP_DIR/passwd.txt /etc/passwd 2>/dev/null
cp $BACKUP_DIR/group.txt /etc/group 2>/dev/null
cp $BACKUP_DIR/ntp.conf.txt /etc/ntp.conf 2>/dev/null
cp $BACKUP_DIR/profile.txt /etc/profile 2>/dev/null

# Restore init scripts
cp $BACKUP_DIR/init.d/cloudflared.txt /etc/init.d/cloudflared 2>/dev/null
chmod +x /etc/init.d/cloudflared 2>/dev/null
cp $BACKUP_DIR/init.d/dns-setup.txt /etc/init.d/dns-setup 2>/dev/null
chmod +x /etc/init.d/dns-setup 2>/dev/null

# Restore network scripts
cp $BACKUP_DIR/network/fix-dns.txt /etc/network/if-pre-up.d/fix-dns 2>/dev/null
chmod +x /etc/network/if-pre-up.d/fix-dns 2>/dev/null

echo "Restore complete! Run: /etc/init.d/cloudflared restart"

=== /media/TCI/ifplugdnet.sh (yang dipakai ifplugd) ===
#!/bin/bash
limit_const_gps=true
#gpsd patch changes should be done one time only after reboot, not during network link up down, so need to shield it

gpsdver=$(gpsd -V)
if [[ $gpsdver == *"3.15"* ]]; then
  cp /media/tci/Patches/gpsd/update323/libgps.so.29 /usr/lib
  killall csmsd.elf
  XMLFILE=/media/TCI/csms/etc/CSMSConfig++.xml
  if [ -f "$XMLFILE" ]
  then
	line=$(grep "nodeId" $XMLFILE | sed -e 's/<nodeId>\(.*\)<\/nodeId>/\1/')
	if [ "$line" == "" ]
	then
		killall gpsd
		killall ntpd
                if [ $limit_const_gps == true ]
                then
                  echo -n -e '$PMTK353,1,0,0,0,0*2A\r\n' > /dev/ttyPS1
                fi
		cp /media/tci/Patches/gpsd/update323/ntp.conf /etc
		ipcrm -a
		cp /media/tci/Patches/gpsd/update323/gpsd /bin
		gpsd -f 8N1 -s 9600 -n /dev/ttyPS1
		ntpd -gGNc /etc/ntp.conf
                echo -e "no df- old gpsd detected, updating...\n"
	else
		if [ $line == "0" ]
		then
			killall gpsd
			killall ntpd
                        if [ $limit_const_gps == true ]
                        then
                          echo -n -e '$PMTK353,1,0,0,0,0*2A\r\n' > /dev/ttyPS1
                        fi
			cp /media/tci/ntpconf.master /media/tci/ntp.conf 
			cp /media/tci/Patches/gpsd/update323/ntp.conf /etc
			ipcrm -a
                        cp /media/tci/Patches/gpsd/update323/gpsd /bin
			gpsd -f 8N1 -s 9600 -n /dev/ttyPS1
			ntpd -gGNc /etc/ntp.conf
                        echo -e "df master - old gpsd detected, updating...\n"
			echo "node $line"
		else
                        echo -e "node $line \n" 
		fi
	fi
  fi
fi
echo -e "ifplugd link detected\n"
# disable ftp
killall inetd
cp -fp /media/TCI/inetd.conf /etc
 
/usr/sbin/inetd
ifdown -a 
sleep 2
sysctl -e -p /etc/sysctl.conf >/dev/null 2>&1
ifup -a
/etc/rc5.d/S99csmsd start
